Met_FirstWeek <- met2[met2$day >= 1 & met2$day <= 7,]
Met_FirstWeek <- Met_FirstWeek[order(Met_FirstWeek$day),]

Station_Means <- aggregate(Met_FirstWeek[c("temp","rh","wind.sp","vis.dist","dew.point","lat","lon","elev")],
  by = list(station = Met_FirstWeek$USAFID),
  FUN = mean,
  na.rm= TRUE)

Station_Means$region <- ifelse(Station_Means$lat >= 39.71 & Station_Means$lon <= -98.00, "NW",
                               ifelse(Station_Means$lat < 39.71 & Station_Means$lon <= -98.00, "SW",
                                      ifelse(Station_Means$lat >= 39.71 & Station_Means$lon > -98.00, "NE", "SE")))

Station_Means$Elev_Cat <- ifelse(Station_Means$elev > 252, "high", "low")
